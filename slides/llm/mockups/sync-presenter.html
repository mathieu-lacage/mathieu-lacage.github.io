step 0/13

Input: x, n rows of d numbers, one row per token

Query: Q = x · WQ; we follow its last row, q4 (« capital »)

Key: K = x · WK, the same x with another weight matrix

Transpose: KT has one column per token

q4 · k1: row 4 of Q times column 1 of KT

q4 · k2: same row of Q, next column of KT

q4 · k3

q4 · k4: the last token sees every token, itself included

Softmax on row 4: the 4 scores become weights that sum to 1

Value: V = x · WV

Mix: z4 = 0.45 v1 + 0.10 v2 + 0.10 v3 + 0.35 v4, row 4 of A times V

Rows 1 to 3: same steps, all at once; row i only sees rows ≤ i

Output: O = Z · WO, back to n × d

import numpy as np

def softmax(s):
    e = np.exp(s - s.max(axis=-1, keepdims=True))
    return e / e.sum(axis=-1, keepdims=True)

def attention(x, Wq, Wk, Wv, Wo):
    """One head: x is n x d; Wq, Wk, Wv are d x dh; Wo is dh x d."""
    n, d = x.shape
    Q = x @ Wq
    K = x @ Wk
    S = Q @ K.T / np.sqrt(Wq.shape[1])
    S += np.triu(np.full((n, n), -np.inf), 1)
    A = softmax(S)
    V = x @ Wv
    Z = A @ V
    return Z @ Wo
def attention_manual(x, Wq, Wk, Wv, Wo):
    """The same result, the matrix products written out."""
    n, d = x.shape
    dh = Wq.shape[1]
    out = np.zeros((n, Wo.shape[1]))
    for i in range(n):
        q = x[i] @ Wq
        s = np.empty(i + 1)
        for j in range(i + 1):
            s[j] = q @ (x[j] @ Wk) / np.sqrt(dh)
        a = softmax(s)
        v = x @ Wv
        z = a @ v[:i + 1]
        out[i] = z @ Wo + x[i]
    return out
